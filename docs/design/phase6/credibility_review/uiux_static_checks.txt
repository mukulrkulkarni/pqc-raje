  6| def credibility(C_ab: np.ndarray, d_j: np.ndarray) -> np.ndarray:
  7|     """Compute credibility S(a,b) from concordance and discordance.
  8|
  9|     Formula:
 10|         S(a,b) = C(a,b) × ∏ F_j(a,b)
 11|
 12|     Where:
 13|         F_j(a,b) = 1                         if d_j(a,b) <= C(a,b)
 14|         F_j(a,b) = (1 - d_j(a,b))/(1 - C(a,b))  otherwise
 15|
 16|     Edge cases:
 17|         - If any d_j >= 1: S = 0 (full veto, including when C(a,b) = 1)
 18|         - If C(a,b) = 1 and all finite d_j < 1: S = 1
 19|
 20|     Args:
 21|         C_ab: Global concordance values, shape (B1, B2).
 22|         d_j: Discordance values per criterion, shape (B1, B2, C).
 23|
 24|     Returns:
 25|         Credibility values, shape (B1, B2).
 26|     """
 27|     S = C_ab.copy()
 28|
 29|     # For each criterion, apply the discordance reduction
 30|     n_criteria = d_j.shape[2]
 31|     for j in range(n_criteria):
 32|         d = d_j[:, :, j]
 33|
 34|         # Full veto (d=1) always sets S=0
 35|         S[d >= 1.0] = 0.0
 36|
 37|         # Where d > C and d < 1: multiply by (1-d)/(1-C)
 38|         mask = (d > C_ab) & (d < 1.0)
 39|         if mask.any():
 40|             denom = 1.0 - C_ab[mask]
 41|             denom = np.where(denom == 0, 1e-10, denom)
 42|             factor = (1.0 - d[mask]) / denom
 43|             S[mask] *= np.clip(factor, 0.0, 1.0)
 44|
 45|     return np.clip(S, 0.0, 1.0)
